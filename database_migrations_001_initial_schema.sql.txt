-- 2. TABEL: CONSULTATIONS
CREATE TABLE IF NOT EXISTS consultations (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW() NOT NULL,
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW() NOT NULL,
  client_name VARCHAR(255) NOT NULL,
  client_email VARCHAR(255) NOT NULL,
  client_whatsapp VARCHAR(50) NOT NULL,
  company_name VARCHAR(255),
  project_type project_type_enum NOT NULL DEFAULT 'custom_web_app',
  target_timeline VARCHAR(100),
  budget_estimate NUMERIC(15, 2) NOT NULL DEFAULT 0,
  down_payment_estimate NUMERIC(15, 2) NOT NULL DEFAULT 0,
  project_features JSONB DEFAULT '[]'::jsonb,
  requirements_note TEXT,
  status consultation_status_enum DEFAULT 'new' NOT NULL,
  ip_address VARCHAR(45),
  user_agent TEXT,
  CONSTRAINT valid_email CHECK (client_email ~* '^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}$')
);

CREATE INDEX IF NOT EXISTS idx_consultations_email ON consultations(client_email);
CREATE INDEX IF NOT EXISTS idx_consultations_status ON consultations(status);
CREATE INDEX IF NOT EXISTS idx_consultations_created_at ON consultations(created_at DESC);

-- 3. TABEL: ESTIMATE_PROPOSALS
CREATE TABLE IF NOT EXISTS estimate_proposals (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  proposal_number VARCHAR(100) UNIQUE NOT NULL,
  consultation_id UUID REFERENCES consultations(id) ON DELETE SET NULL,
  total_amount NUMERIC(15, 2) NOT NULL,
  dp_amount NUMERIC(15, 2) NOT NULL,
  remaining_amount NUMERIC(15, 2) NOT NULL,
  selected_tech_stack JSONB DEFAULT '[]'::jsonb,
  warranty_period_months INT DEFAULT 3,
  valid_until DATE NOT NULL DEFAULT (CURRENT_DATE + INTERVAL '14 days'),
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW() NOT NULL
);

CREATE INDEX IF NOT EXISTS idx_estimate_proposals_number ON estimate_proposals(proposal_number);

-- 4. TABEL: TRANSACTIONS
CREATE TABLE IF NOT EXISTS transactions (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  trx_reference VARCHAR(100) UNIQUE NOT NULL,
  estimate_proposal_id UUID REFERENCES estimate_proposals(id) ON DELETE SET NULL,
  consultation_id UUID REFERENCES consultations(id) ON DELETE SET NULL,
  amount NUMERIC(15, 2) NOT NULL CHECK (amount > 0),
  payment_method payment_method_enum NOT NULL DEFAULT 'QRIS',
  payment_status payment_status_enum NOT NULL DEFAULT 'pending',
  gateway_provider VARCHAR(50) DEFAULT 'midtrans',
  gateway_transaction_id VARCHAR(255),
  qris_payload_url TEXT,
  paid_at TIMESTAMP WITH TIME ZONE,
  receipt_email VARCHAR(255),
  email_sent_at TIMESTAMP WITH TIME ZONE,
  metadata JSONB DEFAULT '{}'::jsonb,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW() NOT NULL
);

CREATE INDEX IF NOT EXISTS idx_transactions_reference ON transactions(trx_reference);
CREATE INDEX IF NOT EXISTS idx_transactions_status ON transactions(payment_status);
