-- 1. TABEL: AUDIT_LOGS
CREATE TABLE IF NOT EXISTS audit_logs (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  event_name VARCHAR(100) NOT NULL,
  payload JSONB NOT NULL,
  ip_address VARCHAR(45),
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW() NOT NULL
);

CREATE INDEX IF NOT EXISTS idx_audit_logs_event ON audit_logs(event_name);
CREATE INDEX IF NOT EXISTS idx_audit_logs_created_at ON audit_logs(created_at DESC);

-- 2. PENGETATAN KEAMANAN ROW LEVEL SECURITY (RLS)
-- Menghapus seluruh policy publik terbuka (Mencegah pencurian data kontak / nominal transaksi)
ALTER TABLE consultations ENABLE ROW LEVEL SECURITY;
ALTER TABLE estimate_proposals ENABLE ROW LEVEL SECURITY;
ALTER TABLE transactions ENABLE ROW LEVEL SECURITY;
ALTER TABLE audit_logs ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Public anonymous insert consultations" ON consultations;
DROP POLICY IF EXISTS "Allow public insert consultations" ON consultations;
DROP POLICY IF EXISTS "Public read estimate proposals" ON estimate_proposals;
DROP POLICY IF EXISTS "Public read transaction status" ON transactions;
DROP POLICY IF EXISTS "Allow public read transaction by reference" ON transactions;

-- A. Consultations: Publik HANYA boleh melakukan INSERT data briefing baru.
-- Publik sama sekali TIDAK BISA membaca (SELECT) data klien lain.
CREATE POLICY "Public anonymous insert consultations" 
  ON consultations FOR INSERT 
  TO anon, authenticated 
  WITH CHECK (true);

-- B. Estimate Proposals: Hanya backend (service_role) yang memiliki akses penuh.
CREATE POLICY "Service role full access estimate_proposals" 
  ON estimate_proposals FOR ALL 
  TO service_role 
  USING (true) 
  WITH CHECK (true);

-- C. Transactions: Seluruh modifikasi status & pencatatan HANYA lewat backend API.
CREATE POLICY "Service role full access transactions" 
  ON transactions FOR ALL 
  TO service_role 
  USING (true) 
  WITH CHECK (true);

-- D. Audit Logs: Hanya internal server
CREATE POLICY "Service role full access audit_logs" 
  ON audit_logs FOR ALL 
  TO service_role 
  USING (true) 
  WITH CHECK (true);
